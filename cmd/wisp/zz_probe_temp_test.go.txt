func TestZZProbeTempSchemas(t *testing.T) {
	// Which tool schema fails to round-trip through json.RawMessage?
	bad := 0
	for _, raw := range probeSchemaSamples() {
		var v any
		if err := json.Unmarshal([]byte(raw), &v); err != nil {
			fmt.Printf("PROBE BAD raw=%q err=%v\n", raw, err)
			bad++
			continue
		}
		re, err := json.Marshal(json.RawMessage(raw))
		if err != nil {
			fmt.Printf("PROBE REMARSHAL-ERR raw=%q err=%v\n", raw, err)
			bad++
			continue
		}
		fmt.Printf("PROBE OK rawlen=%d remarshal=%s\n", len(raw), re)
	}
	fmt.Printf("PROBE bad=%d\n", bad)
}

}
