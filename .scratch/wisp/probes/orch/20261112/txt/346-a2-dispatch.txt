```powershell
$pass = @($c | Select-String '^--- PASS: (\S+)' | ForEach-Object { $_.Matches[0].Groups[1].Value } | Where-Object { $_ -notmatch '/' })
$fail = @($c | Select-String '^--- FAIL: (\S+)' | ForEach-Object { $_.Matches[0].Groups[1].Value } | Where-Object { $_ -notmatch '/' })
$skip = @($c | Select-String '^--- SKIP: (\S+)' | ForEach-Object { $_.Matches[0].Groups[1].Value } | Where-Object { $_ -notmatch '/' })
Write-Output "RUN=$($runs.Count) PASS=$($pass.Count) FAIL=$($fail.Count) SKIP=$($skip.Count)"
Write-Output "FAIL names: $($fail -join ', ')"
Write-Output "SKIP names: $($skip -join ', ')"
```
