		// not hold the handle. The refusal is a return value, so it is readable
		// here and on the audit line below, not only in a log the engine prints.
		closeErr = scope.Close()
	}
	b.log("tools: C25 scope closed task=%s was_open=%v dropped=%d open_scopes=%d close_err=%v",
		taskID, open, dropped, left, closeErr)
}

// assessorFor returns the assessor whose R4 input is bound to this task's
// scope. The frozen C19 TaintDetector seam takes no tool name
// (internal/risk/assessor.go:165-171), so the binding is per scope, hence per
// task; the canonicalizer and classifier are shared and immutable.
//
// The `taskID == ""` limb is on purpose, and it UNDER-REPORTS rather than refuses:
// an empty task id falls through to `return b.assess`, which holds no TaintDetector,
// so ruleTaint short-circuits on its `if ctx.taint == nil` guard
// (internal/risk/rules_gateway.go) and the R4 branch never fires for that call - no
// error, no trace. Read the next sentence as scope, not as coverage: the guard only picks
// WHICH assessor answers the one `verdict := b.assessorFor(req.TaskID).Assess(...)` line, and
// the fallback b.assess is a full assessor minus the detector (New builds it with the
// canonicalizer and the sensitive classifier), so the other rules are STRUCTURALLY still
// reached on that path - and nothing in ticket 309 asserts it. This file instruments one
// fact: whether a detector is attached. The other gates belong to each rule's own teeth
// tests (the two ticket-236 carriers that really feed an empty TaskID into Execute assert
// the refusal, not the rules); only R4 is dropped here, silently.
//
// Two things keep that sentence honest, both checkable now:
//   - No production caller reaches the limb today. Dispatch funnels through the one
//     `newTaskID()` construction point in internal/agent/loop.go (id 36 chars, never empty);
//     Test309ToolRequestProductionConstructionNail counts every production construction
//     point of that type across the four roots cmd, internal, tools and scripts from the AST
//     and pins the TOTAL to 1, named as that file - so a second one rings wherever it is
//     written, including inside internal/agent, which the pre-309-r2 version skipped whole.
//   - Removing the limb to look stricter is not correctness: scopeMarks in
//     internal/risk/provenance.go fail-closes an unregistered scope to R4 whenever a
//     DIFFERENT scope still holds marks, so with the limb gone an empty id trades this
//     silent under-report for a conditional false-positive storm, not for a check.
//
// The other two limbs of that guard - `b.injected` and `b.prov == nil` - are the
// assembly-is-wrong family, and NEITHER has an instrument that names it. Measured at HEAD
// 6f389389, 2026-10-11 14:4x +08, whole package (ruler and readings:
// .scratch/wisp/probes/309/r2/logs/prov-rigs.stdout): striking `b.injected` out leaves it
// rc=0 - nothing rings anywhere - while striking `b.prov == nil` out leaves it rc=1 on 16
// top-level tests that are
// plumbing noise (a nil *Provenance reaching Detector(), calls answering L2), not a test about
// that limb; 309-v1's rc=0 pair was scoped to the 309 instrument, which stays green for both.
// `injected` is inert in production because nothing writes the Options field named Assessor
// (write-point census R359-4, .scratch/wisp/probes/309/r2/30-comments.md). Ticket 309 AC#0
// ruled shape B and refused a second fixture for this family, so these lines are the register,
// not a guarantee: whoever first wires that field - or relies on prov being non-nil - owns the
// test that has to ring.
//
// Who rings if this limb's shape changes is Test309AssessorForEmptyTaskIDDropsTaintDetector:
// it fails if a weakened guard attaches the detector for the empty id, and fails if the
// attach branch stops attaching the detector for a real id. The compiler guards nothing
// here - that test is the only instrument, and this comment states the debt, not a guarantee.
func (b *Bridge) assessorFor(taskID string) *risk.RiskAssessor {
	// `taskID == ""` here is deliberate and under-reports (R4 silently dropped for
	// this call); see the doc block above and Test309AssessorForEmptyTaskIDDropsTaintDetector.
	if b.injected || b.prov == nil || taskID == "" {
		return b.assess
	}
	return risk.NewRiskAssessor().
		WithCanonicalizer(b.paths).
		WithSensitiveClassifier(b.classifier).
		WithTaintDetector(b.prov.Detector(taskID))
}
