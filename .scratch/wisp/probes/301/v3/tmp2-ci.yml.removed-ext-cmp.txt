concurrency:
  # Q-25 (owner 2026-09-23, approved "加"): push events get a per-sha group.
  # With a per-ref group and cancel-in-progress=false, GitHub keeps only the
  # OLDEST queued run once the in-progress one finishes and cancels the rest,
  # so a burst of pushes permanently destroys the step-level logs of the
  # dropped commits - and those logs are our only evidence for several gates.
  # Measured cost of the old shape (Q-25 evidence, 09-21 evening): 4 of the 6
  # most recent dev runs were superseded this way.
  # PR events deliberately keep the old per-ref group + cancel-in-progress so
  # a force-pushed branch still supersedes its own previous run.
  group: ci-${{ github.workflow }}-${{ github.ref }}${{ github.event_name == 'push' && format('-{0}', github.sha) || '' }}
  cancel-in-progress: ${{ github.event_name == 'pull_request' }}
