package winsec

import (
	"fmt"
	"io/fs"
	"os"
	"path/filepath"
	"syscall"
)

func openPrivatePinned(path string, flags int, perm fs.FileMode) (*os.File, error) {
	parent := filepath.Dir(path)
	pinned, err := pinPlacement(parent)
	if err != nil {
		return nil, err
	}
	defer pinned.Close()
	lfd, err := syscall.Openat(int(pinned.Fd()), filepath.Base(path), flags|syscall.O_NOFOLLOW, uint32(perm.Perm()))
	if err != nil {
		return nil, &os.PathError{Op: "open", Path: path, Err: fmt.Errorf("on the pinned parent: %w", err)}
	}
	return os.NewFile(uintptr(lfd), path), nil
}
