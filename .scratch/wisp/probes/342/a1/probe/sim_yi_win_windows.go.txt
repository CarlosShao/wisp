package winsec

import (
	"fmt"
	"io/fs"
	"os"
	"path/filepath"
)

func openPrivateAt(path string, flags int, perm fs.FileMode) (*os.File, error) {
	comps := pathComponents(path)
	if len(comps) < 2 {
		return nil, &os.PathError{Op: "open", Path: path, Err: fmt.Errorf("the pinning walk expects a rooted spelling, got %q", path)}
	}
	root, err := os.OpenRoot(filepath.VolumeName(path) + string(os.PathSeparator))
	if err != nil {
		return nil, &os.PathError{Op: "open", Path: path, Err: fmt.Errorf("the walk could not open the volume root: %w", err)}
	}
	defer root.Close()
	cur := root
	for _, comp := range comps[:len(comps)-1] {
		next, err := cur.OpenRoot(comp)
		if err != nil {
			return nil, &os.PathError{Op: "open", Path: path, Err: fmt.Errorf("the walk could not pin the ancestor %q: %w", comp, err)}
		}
		if cur != root {
			_ = cur.Close()
		}
		cur = next
	}
	defer cur.Close()
	return cur.OpenFile(filepath.Base(path), flags, perm)
}
