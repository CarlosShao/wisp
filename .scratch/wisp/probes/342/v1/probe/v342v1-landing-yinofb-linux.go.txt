// v342v1 MIS-SHAPE POSITIVE CONTROL (the "yinofb" row) — exported trees only, never
// tracked. This is deliberately the WRONG landing: it pins the parent like 乙 but then
// forces O_EXCL on the leaf, so PrivateFile refuses a name that already exists.
//
// Purpose = ruler sensitivity. A shape that quietly breaks the overwrite promise must
// turn real red, otherwise a green "top-red roster = 0" for the good landings means
// nothing. Expected victims (342-a1 reported 4): the ticket 322 leaf-race pin, the
// ticket 322 overwrite pin, and two of the sealhandle privateroute pins.
package winsec

import (
	"fmt"
	"io/fs"
	"os"
	"path/filepath"
	"syscall"
)

func openPrivateAt(path string, flags int, perm fs.FileMode) (*os.File, error) {
	parent := filepath.Dir(path)
	pinned, err := pinPlacement(parent)
	if err != nil {
		return nil, err
	}
	defer pinned.Close()
	lfd, err := syscall.Openat(int(pinned.Fd()), filepath.Base(path),
		flags|syscall.O_EXCL|syscall.O_CREAT|syscall.O_NOFOLLOW, uint32(perm.Perm()))
	if err != nil {
		return nil, &os.PathError{Op: "open", Path: path, Err: fmt.Errorf("v342v1 mis-shape on the pinned parent: %w", err)}
	}
	return os.NewFile(uintptr(lfd), path), nil
}
