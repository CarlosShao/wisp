A. 突变台脚本 `/d/tmp/324v1b/scripts/mutate324.py`（逐字）

    import hashlib
    import os
    import sys

    REL = "cmd/wisp/config_receipt_255_test.go"

    DROP_OLD = (
        "\t\tif sectionMembers[identName(rest)] == false {\n"
        "\t\t\tcontinue\n"
        "\t\t}\n"
    )
    DROP_NEW = "\t\t/* MUTATION 324-v1b dropgate: schema-member evidence removed */\n"

    BLIND_OLD = (
        "\tout := map[string]bool{}\n"
        "\tft := reflect.TypeOf(config.Config{})\n"
    )
    BLIND_NEW = (
        "\tout := map[string]bool{}\n"
        "\tif true {\n"
        "\t\treturn out /* MUTATION 324-v1b blind: member reading pinned empty */\n"
        "\t}\n"
        "\tft := reflect.TypeOf(config.Config{})\n"
    )


    def sha(path):
        with open(path, "rb") as fh:
            return hashlib.sha256(fh.read()).hexdigest()


    def apply(tree, old, new, tag):
        path = os.path.join(tree, REL)
        before = sha(path)
        with open(path, "r", newline="", encoding="utf-8") as fh:
            text = fh.read()
        n = text.count(old)
        if n != 1:
            print("ABORT %s: anchor occurs %d times (want 1)" % (tag, n))
            sys.exit(3)
        with open(path, "w", newline="", encoding="utf-8") as fh:
            fh.write(text.replace(old, new))
        print("MUTATE %s tree=%s sha_before=%s sha_after=%s" % (tag, tree, before, sha(path)))


    def restore(tree, pristine):
        path = os.path.join(tree, REL)
        before = sha(path)
        with open(pristine, "rb") as src, open(path, "wb") as dst:
            dst.write(src.read())
        a, b = sha(path), sha(pristine)
        print("RESTORE tree=%s sha_before=%s sha_after=%s pristine=%s match=%s"
              % (tree, before, a, b, "YES" if a == b else "NO"))
        sys.exit(0 if a == b else 4)


    if __name__ == "__main__":
        mode = sys.argv[1]
        if mode == "dropgate":
            apply(sys.argv[2], DROP_OLD, DROP_NEW, "dropgate")
        elif mode == "blind":
            apply(sys.argv[2], BLIND_OLD, BLIND_NEW, "blind")
        elif mode == "restore":
            restore(sys.argv[2], sys.argv[3])
        elif mode == "sha":
            print("SHA tree=%s file=%s sha=%s" % (sys.argv[2], REL, sha(os.path.join(sys.argv[2], REL))))

⚠ 两枚台面事故具名（都⛔ 计入判据，都⛔ 删件）：
 ① 第一版 `apply()` 用 `open(path,"r",newline="")` ⛔ 指 encoding ⇒ Windows 默认 `gbk` 解 UTF-8 源 ⇒
    `UnicodeDecodeError: 'gbk' codec can't decode byte 0x8d in position 80`，脚本在**读**阶段就退 3，
    盘上⛔ 没改 ⇒ 我随后那一发 `go test -c` 编的是**未突变**的源、跑出 `rc=0 / 3 枚顶层 PASS`
    （件＝`/d/tmp/324v1b/mutA-2-dropgate.log`）。那枚绿⛔ 是"突变后仍绿"的读数＝假读数。
    定式＝**突变台件的脚本自己也要一发正控**：跑完必须回读 `sha_after`，⛔ 看测试色就认定突变落了盘。
    改正＝两处 `open` 都带 `encoding="utf-8"`；重跑 ⇒ `sha_after=4ada00e6…af6e`＋三枚红（读数件 02 的 `mutA-3-dropgate-real`）。
 ② 台件脚本用 `printf '----- xxx -----\n'` 起头打了三枚分隔行 ⇒ bash 把开头的 `--` 当选项
    （`printf: --: invalid option`），那三枚标记⛔ 进了文件（正文没丢）。本件的标记是手写的。
