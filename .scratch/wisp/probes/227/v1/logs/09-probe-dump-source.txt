// 227-v1 终裁腿专用探针：⛔ 入库、⛔ 是判据，只打在仓外导出树的物理字节上。
// 目的＝量"那三句日志字面串在它们各自被打印的那一发上是否说实话"。

import (
	"strings"
	"testing"

// 世界 A＝P4（删 map 条目，本腿实测不补回）⇒ 走的是 diverged 那支＝句子③打印。
func TestV1ProbeLogInMapDeleteWorld(t *testing.T) {
	logs := captureSlog(t)
	start := matrixBase + `
[llm]
[llm.providers.deepseek]
base_url = "https://api.deepseek.com/v1"
`
	m, path, mutate := newTestManager(t, start)
	noTable := strings.Replace(start,
		"\n[llm]\n[llm.providers.deepseek]\nbase_url = \"https://api.deepseek.com/v1\"\n", "\n", 1)
	mutate(noTable)
	if err := m.AddAllowedDir(testDir); err != nil {
		t.Fatalf("AddAllowedDir: %v", err)
	}
	got := readLoaded(t, path)
	_, restored := got.LLM.Providers["deepseek"]
	text := logs.String()
	t.Logf("PROBE-A providers-restored=%v", restored)
	t.Logf("PROBE-A sentence3-kept-branch-printed=%v", strings.Contains(text, "kept hand edits this process does not hold"))
	t.Logf("PROBE-A sentence3-delete-claim-printed=%v",
		strings.Contains(text, "a key line the operator deleted by hand comes back at its schema default value"))
	t.Logf("PROBE-A file-still-mentions-deepseek=%v", strings.Contains(readText(t, path), "deepseek"))
	t.Logf("PROBE-A file-has-base_url-line=%v", strings.Contains(readText(t, path), "base_url"))
}

// 世界 B＝P1（删非默认标量行）⇒ 同样走 diverged 支＝句子③打印，且这一次它说的是实话。
func TestV1ProbeLogInScalarDeleteWorld(t *testing.T) {
	logs := captureSlog(t)
	start := strings.Replace(matrixBase, "size = 56", "size = 60  # my size", 1)
	m, path, mutate := newTestManager(t, start)
	mutate(strings.Replace(start, "size = 60  # my size\n", "", 1))
	if err := m.AddAllowedDir(testDir); err != nil {
		t.Fatalf("AddAllowedDir: %v", err)
	}
	text := logs.String()
	body := readText(t, path)
	t.Logf("PROBE-B size-line-back=%v", strings.Contains(body, "size = 56"))
	t.Logf("PROBE-B comment-gone=%v", !strings.Contains(body, "my size"))
	t.Logf("PROBE-B sentence3-delete-claim-printed=%v",
		strings.Contains(text, "a key line the operator deleted by hand comes back at its schema default value"))
}

// 世界 C＝句子①在 schema 落后那一支上到底有没有被打印（而那一世界文件字节确实被迁移改了）。
func TestV1ProbeLogInStaleSchemaWorld(t *testing.T) {
	logs := captureSlog(t)
	hand := strings.Replace(matrixBase,
		"allowed_dirs = []", `allowed_dirs = ["D:\\data"]  # hand set`, 1)
	noVersion := strings.Replace(hand, "schema_version = 2\n", "", 1)
	m, path, mutate := newTestManager(t, matrixBase)
	mutate(noVersion)
	before := readText(t, path)
	mb := statMod(t, path)
	if err := m.AddAllowedDir(testDir); err != nil {
		t.Fatalf("AddAllowedDir: %v", err)
	}
	text := logs.String()
	moved := readText(t, path) != before || !statMod(t, path).Equal(mb)
	t.Logf("PROBE-C sentence1-printed=%v", strings.Contains(text, "already carries this key"))
	t.Logf("PROBE-C sentence1-stays-put-clause-printed=%v",
		strings.Contains(text, "the file's bytes and mtime stayed put only because"))
	t.Logf("PROBE-C bytes-or-mtime-actually-moved=%v", moved)
	t.Logf("PROBE-C comment-survived=%v", strings.Contains(readText(t, path), "hand set"))
}
