func FloatToPCM16(in []float32) []int16 {
	out := make([]int16, len(in))
	for i, v := range in {
		s := int64(v*32767 + 0.5)
		if v < 0 {
			s = int64(v*32768 - 0.5)
		}
		out[i] = clampI16(int32(s))
	}
	return out
