func (l *Loop) Steer(text string) bool {
	if !l.opt.Config.SteeringEnabled || strings.TrimSpace(text) == "" {
		return false
	}
	l.mu.Lock()
	defer l.mu.Unlock()
	l.steer = append(l.steer, userMessage(text))
	return true
}
